cmake_minimum_required(VERSION 3.16)
project(mobile_manipulator_control)
find_package(ament_cmake REQUIRED)
foreach(dep hardware_interface pluginlib rclcpp rclcpp_lifecycle sensor_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(unity_arm_system SHARED src/unity_arm_system.cpp)
target_compile_features(unity_arm_system PUBLIC cxx_std_17)
target_compile_options(unity_arm_system PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(unity_arm_system PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(unity_arm_system hardware_interface pluginlib rclcpp rclcpp_lifecycle sensor_msgs)
pluginlib_export_plugin_description_file(hardware_interface unity_arm_plugin.xml)
find_package(nlohmann_json REQUIRED)
add_executable(unity_control_endpoint src/unity_control_endpoint.cpp)
target_compile_features(unity_control_endpoint PUBLIC cxx_std_17)
target_compile_options(unity_control_endpoint PRIVATE -Wall -Wextra -Wpedantic -O2)
target_include_directories(unity_control_endpoint PRIVATE include)
target_link_libraries(unity_control_endpoint rclcpp::rclcpp nlohmann_json::nlohmann_json)
find_package(controller_manager_msgs REQUIRED)
find_package(lifecycle_msgs REQUIRED)
add_executable(arm_recovery_supervisor src/arm_recovery_supervisor.cpp)
target_compile_features(arm_recovery_supervisor PUBLIC cxx_std_17)
target_compile_options(arm_recovery_supervisor PRIVATE -Wall -Wextra -Wpedantic -O2)
target_include_directories(arm_recovery_supervisor PRIVATE include)
target_link_libraries(arm_recovery_supervisor rclcpp::rclcpp ${controller_manager_msgs_TARGETS}
  ${lifecycle_msgs_TARGETS} ${sensor_msgs_TARGETS})
find_package(std_msgs REQUIRED)
add_executable(control_description src/control_description.cpp)
target_compile_features(control_description PUBLIC cxx_std_17)
target_compile_options(control_description PRIVATE -Wall -Wextra -Wpedantic -O2)
target_link_libraries(control_description rclcpp::rclcpp ${std_msgs_TARGETS})
install(TARGETS unity_control_endpoint arm_recovery_supervisor control_description
  RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS unity_arm_system LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
# Experiment tools (simulation only).
foreach(dep ament_index_cpp control_msgs geometry_msgs rclcpp_action rosgraph_msgs tf2_msgs trajectory_msgs urdf)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(measure_transport src/measure_transport.cpp)
target_link_libraries(measure_transport rclcpp::rclcpp ${rosgraph_msgs_TARGETS} ${sensor_msgs_TARGETS}
  nlohmann_json::nlohmann_json)
add_executable(arm_experiment src/arm_experiment.cpp)
target_link_libraries(arm_experiment rclcpp::rclcpp rclcpp_action::rclcpp_action ${control_msgs_TARGETS}
  ${geometry_msgs_TARGETS} ${sensor_msgs_TARGETS} ${trajectory_msgs_TARGETS} urdf::urdf
  ament_index_cpp::ament_index_cpp nlohmann_json::nlohmann_json)
add_executable(base_step_test src/base_step_test.cpp)
target_link_libraries(base_step_test rclcpp::rclcpp ${geometry_msgs_TARGETS} ${sensor_msgs_TARGETS}
  ${tf2_msgs_TARGETS} nlohmann_json::nlohmann_json)
foreach(tool measure_transport arm_experiment base_step_test)
  target_compile_features(${tool} PRIVATE cxx_std_17)
  target_compile_options(${tool} PRIVATE -Wall -Wextra -Wpedantic -O2)
  target_include_directories(${tool} PRIVATE include)
endforeach()
install(TARGETS measure_transport arm_experiment base_step_test RUNTIME DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_joint_packet test/test_joint_packet.cpp)
  target_include_directories(test_joint_packet PRIVATE include)
  ament_add_gtest(test_tcp_protocol test/test_tcp_protocol.cpp)
  target_include_directories(test_tcp_protocol PRIVATE include)
  target_link_libraries(test_tcp_protocol nlohmann_json::nlohmann_json)
  ament_add_gtest(test_feedback_freshness test/test_feedback_freshness.cpp)
  target_include_directories(test_feedback_freshness PRIVATE include)
  ament_add_gtest(test_feedback_check test/test_feedback_check.cpp)
  target_include_directories(test_feedback_check PRIVATE include)
endif()
ament_export_include_directories(include)
ament_package()
